\begin{simbolos}
  \item[$ A \circ B $] Função composta de B em A
  \item[$ \underset{A_1, A_2 ... A_n}{argmin}$] Função que minimiza o resultado de outra função através da escolha dos parâmetros $A_1$,$A_2$... $A_n$
  \item[$ \left\|\; \right\|^2 $] Módulo de um vetor 
  \item[$ A\to B$] Transformação de A para B
  \item[$ \psi $] Letra grega minúscula psi
  \item[$ \phi$] Letra grega minúscula phi
  \item[$\sigma$] Letra grega minúscula sigma
  \item[$\Gamma$] Letra grega maiúscula gamma
  \item[$\gamma$] Letra grega minúscula gamma
  \item[$\mu$] Letra grega minúscula mu
  \item[$\nu$] Letra grega minúscula nu
  \item[$\mathbb {R}$] Conjunto dos Números Reais
  \item[$\int $] Integral
\end{simbolos}
